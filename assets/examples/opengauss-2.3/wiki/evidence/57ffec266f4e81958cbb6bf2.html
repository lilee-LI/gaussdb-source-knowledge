static int gimme_pool_size(int nr_rel)
{
    double size;
    int minsize;
    int maxsize;

    /* Legal pool size *must* be at least 2, so ignore attempt to select 1 */
    if (u_sess->attr.attr_sql.Geqo_pool_size >= 2)
        return u_sess->attr.attr_sql.Geqo_pool_size;

    size = pow(2.0, nr_rel + 1.0);

    maxsize = 50 * u_sess->attr.attr_sql.Geqo_effort; /* 50 to 500 individuals */
    if (size > maxsize) {
        return maxsize;
    }

    minsize = 10 * u_sess->attr.attr_sql.Geqo_effort; /* 10 to 100 individuals */
    if (size < minsize) {
        return minsize;
    }

    return (int)ceil(size);
}

/*
 * Return either configured number of generations or a good default
 *
 * The default is the same as the pool size, which allows us to be
 * sure that less-fit individuals get pushed out of the breeding
 * population before the run finishes.
 */
static int gimme_number_generations(int pool_size)
{
    if (u_sess->attr.attr_sql.Geqo_generations > 0)
        return u_sess->attr.attr_sql.Geqo_generations;

    return pool_size;
}
